cmake_minimum_required(VERSION 3.24)
project(KiloInstaller LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

function(kilo_target target)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /permissive- /EHsc)
  target_link_libraries(${target} PRIVATE d2d1 dwrite dwmapi shell32 ole32 setupapi bcrypt advapi32)
  target_link_options(${target} PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest)
endfunction()

add_executable(KiloUninstall WIN32 ui.cpp engine.cpp icon.rc)
target_compile_definitions(KiloUninstall PRIVATE KILO_UNINSTALL)
kilo_target(KiloUninstall)

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/payload.cab")
  add_executable(KiloSetup WIN32 ui.cpp engine.cpp icon.rc payload.rc)
  target_compile_definitions(KiloSetup PRIVATE KILO_PAYLOAD)
  kilo_target(KiloSetup)
  add_executable(KiloInstallerTests tests.cpp engine.cpp)
  target_compile_definitions(KiloInstallerTests PRIVATE KILO_PAYLOAD)
  kilo_target(KiloInstallerTests)
  enable_testing()
  add_test(NAME InstallerLifecycle COMMAND KiloInstallerTests "${CMAKE_CURRENT_SOURCE_DIR}/payload.cab")
endif()
